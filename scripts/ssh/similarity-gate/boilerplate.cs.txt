// ---------------------------------------------------------------------------
// System.IO.Stream —— 不可 Seek 的流必须显式拒绝这三个成员。
// Stream 把 Length/Position 定义成非抽象的虚成员，子类只能 override 并抛异常；
// 没有「不实现」这个选项。
// ---------------------------------------------------------------------------
class StreamNonSeekableBoilerplate
{
    public override bool CanSeek => false;

    public override long Length => throw new NotSupportedException();

    public override long Position
    {
        get => throw new NotSupportedException();
        set => throw new NotSupportedException();
    }

    public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();

    public override void SetLength(long value) => throw new NotSupportedException();
}

class StreamOneDirectionBoilerplate

// ---------------------------------------------------------------------------
// System.IO.Stream —— byte[]/offset/count 重载转发到 Memory<byte> 重载。
// 基类定死了这两组签名，转发写法只有这一种。
// ---------------------------------------------------------------------------
class StreamForwardingBoilerplate
{
    public override Task<int> ReadAsync(byte[] buffer, int offset, int count, CancellationToken cancellationToken)
    {
        ValidateBufferArguments(buffer, offset, count);
        return ReadAsync(buffer.AsMemory(offset, count), cancellationToken).AsTask();
    }

    public override Task WriteAsync(byte[] buffer, int offset, int count, CancellationToken cancellationToken)
    {
        ValidateBufferArguments(buffer, offset, count);
        return WriteAsync(buffer.AsMemory(offset, count), cancellationToken).AsTask();
    }
}

// ---------------------------------------------------------------------------
// IDisposable / IAsyncDisposable —— 带守卫的标准释放模式。
// 这个形状是 .NET 的 Dispose 约定规定的，偏离它才是错的。
// ---------------------------------------------------------------------------
class DisposeBoilerplate
{
    private bool _disposed;

    protected override void Dispose(bool disposing)
    {
        if (_disposed)
        {
            return;
        }
        _disposed = true;
        base.Dispose(disposing);
    }

    public void Dispose()
    {
        if (_disposed)
        {
            return;
        }
        _disposed = true;
        GC.SuppressFinalize(this);
    }
}

// ---------------------------------------------------------------------------
// System.IO.Stream —— 同一组转发的**表达式体**写法。
// 上面那段带 ValidateBufferArguments 与大括号，token 序列与这一种不同；
// 而一个包装流（计量、限速、加解密）里通常两行就写完，没有别的写法可选：
// 签名是基类定死的，方法体是「转调 Memory 重载再 .AsTask()」这一件事。
// ---------------------------------------------------------------------------
class StreamForwardingExpressionBoilerplate
{
    public override Task<int> ReadAsync(
        byte[] buffer, int offset, int count, CancellationToken cancellationToken) =>
        ReadAsync(buffer.AsMemory(offset, count), cancellationToken).AsTask();

    public override Task WriteAsync(
        byte[] buffer, int offset, int count, CancellationToken cancellationToken) =>
        WriteAsync(buffer.AsMemory(offset, count), cancellationToken).AsTask();

    public override Task FlushAsync(CancellationToken cancellationToken) =>
        _inner.FlushAsync(cancellationToken);

    public override void Flush() => _inner.Flush();
}

// System.IO.Stream —— 单向适配器的能力声明。
// 把一个非 Stream 的东西（IBufferWriter、ReadOnlySequence…）接到只认 Stream 的

// 再把 Length/Position/Seek/SetLength 一律拒掉。
// 顺序是 Stream 成员的声明顺序，写法是 C# 表达式体，没有第二种选择。
// ---------------------------------------------------------------------------
class StreamWriteOnlyAdapterBoilerplate
{
    public override bool CanRead => false;

    public override bool CanSeek => false;

    public override bool CanWrite => true;

    public override long Length => throw new NotSupportedException();

    public override long Position
    {
        get => throw new NotSupportedException();
        set => throw new NotSupportedException();
    }
}
